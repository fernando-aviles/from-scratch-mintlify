```javascript Node.js
import express from "express";
import { Webhook } from "@relaybox/node";

const app = express();
const webhook = new Webhook(process.env.RELAYBOX_SIGNING_SECRET);

// Use the raw body: re-serialized JSON will not match the signature.
app.post("/webhooks", express.raw({ type: "application/json" }), (req, res) => {
  try {
    const event = webhook.verify(req.body, req.headers);
    console.log(`Received ${event.type} (${event.id})`);
    res.sendStatus(204);
  } catch (err) {
    // The signature is missing, invalid, or too old.
    res.status(400).send("Invalid signature");
  }
});
```

```python Python
import os
from flask import Flask, request
from relaybox import Webhook, WebhookVerificationError

app = Flask(__name__)
webhook = Webhook(os.environ["RELAYBOX_SIGNING_SECRET"])

@app.post("/webhooks")
def handle_webhook():
    try:
        # Pass the raw bytes, not request.json
        event = webhook.verify(request.get_data(), request.headers)
    except WebhookVerificationError:
        return "Invalid signature", 400

    print(f"Received {event.type} ({event.id})")
    return "", 204
```

```go Go
package main

import (
	"io"
	"log"
	"net/http"
	"os"

	relaybox "github.com/relaybox/relaybox-go/v3"
)

func main() {
	wh := relaybox.NewWebhook(os.Getenv("RELAYBOX_SIGNING_SECRET"))

	http.HandleFunc("/webhooks", func(w http.ResponseWriter, r *http.Request) {
		body, _ := io.ReadAll(r.Body)

		event, err := wh.Verify(body, r.Header)
		if err != nil {
			// The signature is missing, invalid, or too old.
			http.Error(w, "Invalid signature", http.StatusBadRequest)
			return
		}

		log.Printf("Received %s (%s)", event.Type, event.ID)
		w.WriteHeader(http.StatusNoContent)
	})

	log.Fatal(http.ListenAndServe(":3000", nil))
}
```
